<?php
// Στοιχεία σύνδεσης (προεπιλογές XAMPP/WAMP)
$servername = "localhost";
$username   = "root";
$password   = "";
$dbname     = "company";

}

mysqli_set_charset($conn, "utf8mb4");
header("Content-Type: text/html; charset=utf-8");

$email = "papkostas@gmail.com";
$sql = "SELECT FirstName, Lastname FROM Customers WHERE email = '$email'";

// 3. Εμφάνιση ονόματος και επιθέτου με echo

    echo "Δεν βρέθηκε πελάτης με email $email";

25 klasi-roloi PHP

<meta charset="UTF-8">
<?php
date_default_timezone_set('Europe/Athens');

class roloi {
    public $wr;
    public $lep;
    public $deft;

    // Κατασκευαστής: έλεγχος εγκυρότητας, αλλιώς 0
    function __construct($w, $l, $d) {
        $this->wr   = ($w >= 0 && $w <= 23) ? $w : 0;
        $this->lep  = ($l >= 0 && $l <= 59) ? $l : 0;
        $this->deft = ($d >= 0 && $d <= 59) ? $d : 0;
    }

    // Εμφάνιση ω:λ:δ
    function emfanisi() {
        echo $this->wr . ":" . $this->lep . ":" . $this->deft;
    }

    // Συνολικά δευτερόλεπτα
    function deftera() {
        return $this->wr * 3600 + $this->lep * 60 + $this->deft;
    }
}

$r = new roloi((int)date('H'), (int)date('i'), (int)date('s'));
$r->emfanisi();
echo "<br>Δευτερόλεπτα: " . $r->deftera();
?>

26 klasi-metr PHP

<meta charset="UTF-8">
<?php
class metr {
    private $i;                    // ιδιωτική ιδιότητα
    public static $ano  = 10;      // ιδιότητες κλάσης (static)
    public static $kato = 0;

    // Κατασκευαστής: η τιμή μένει μέσα στα όρια
    function __construct($t) {
        if ($t > self::$ano) $this->i = self::$ano;
        elseif ($t < self::$kato) $this->i = self::$kato;
        else $this->i = $t;
    }

    function auxisi() {
        if ($this->i < self::$ano) $this->i++;
    }

    function meiosi() {
        if ($this->i > self::$kato) $this->i--;
    }

    function ektyposi() {
        echo $this->i;
    }
}

$m = new metr(5);
$m->auxisi();
$m->ektyposi(); // 6
?>
